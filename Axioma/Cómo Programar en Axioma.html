01Primeros pasos

Un programa Axioma se guarda con la extensión .axm y se ejecuta con el intérprete de referencia:

python3 run_axioma.py programa.axm
python3 run_axioma.py < programa.axm   -- también acepta stdin

El runner valida la extensión: un archivo .ax (el formato histórico) será rechazado con un mensaje que indica renombrarlo a .axm.

-- hola_mundo.axm
out("¡Hola, Axioma!")

03Variables y tipos

let x := 10
let pi : RR := 3.14159        -- con anotación de tipo
let n :: ZZ                  -- declara el dominio sin inicializar
x += 5
x -= 1
x *= 2
x /= 3

Axioma no tiene if clásico; la forma condicional usa let <condición> begin ... end, encadenable con given (elif) y otherwise (else):

let x > 5 begin
  out("x es mayor que 5")
end given x > 0 begin
  out("x es positivo")
end otherwise begin
  out("x es cero o negativo")
end

05Control de flujo

Bucle each

each i in {1..5} begin
  out(s"i = {i}")
end

Bucle repeat ... until

let count := 0
repeat begin
  count += 1
end until count === 3

Comparación por patrones: given ... case

Distinto del given usado como elif: cuando va seguido de dos puntos (:), es una comparación por patrones contra un valor concreto.

given n:
  case 1 -> out("uno")
  case 2 -> out("dos")
  otherwise -> out("otro")

Cada brazo acepta un bloque begin...end o una sola sentencia, incluso en la línea siguiente — útil para anidar otro given:

given is_null(nodo):
  case T ->
    out("vacío")
  otherwise ->
    out(nodo'valor)

stop termina el bucle actual; next salta a la siguiente iteración; skip no hace nada.

06Funciones y lambdas

where cuadrado(x) begin
  report x * x
end

let doble := fn(x) => x * 2
out(doble(21))             -- 42
out(apply(doble, [1,2,3])) -- [2, 4, 6]

Los parámetros admiten valores por defecto (nombre := valor) y variádicos (...nombre).

08Tipos de Datos Abstractos

Una structure define un TDA con campos (property), un inicializador (create) y métodos (where):

structure Punto :=
  property x
  property y

  create(px, py) begin
    this'x := px
    this'y := py
  end

  where distancia() begin
    report ((this'x ** 2) + (this'y ** 2)) ** 0.5
  end
end

let p := Punto(3, 4)
out(p'distancia())   -- 5

Campos con valor por defecto

structure Nodo :=
  property valor
  property siguiente := Null   -- se inicializa antes de create()
end

Campos static (compartidos entre instancias)

structure Contador :=
  static property total := 0
  create() begin
    Contador'total := Contador'total + 1
  end
end

Contador(); Contador(); Contador()
out(Contador'total)   -- 3 — se accede por el nombre de la clase o por cualquier instancia

Campos fixed / immutable (de solo asignación inicial)

Pueden escribirse durante create(), pero quedan bloqueados después. Reasignarlos lanza un error capturable:

structure Punto(immutable) :=
  fixed property x
  fixed property y
  create(px, py) begin this'x := px; this'y := py end
end

let p := Punto(3, 4)
check begin
  p'x := 99
end
catch as e begin
  out(s"capturado: {e}")   -- Cannot reassign 'x' — declared 'fixed' ...
end
Nota de diseño Un nombre de campo puede coincidir con una palabra reservada (next, prev, type…) siempre que se declare con property: property next := Null. Esto es intencional — son justamente los nombres más naturales para nodos enlazados.

09Nodos, referencias y punteros

Una instancia de structure es un tipo de referencia: asignarla a una variable, o a un campo de otra instancia, no la copia — comparte el mismo objeto en memoria. Por eso nodo'siguiente := otroNodo ya crea el enlace, sin sintaxis adicional de punteros.

Lista enlazada

structure Nodo :=
  property valor
  property siguiente := Null
  create(v) begin this'valor := v end
end

let cabeza := Nodo(1)
cabeza'siguiente := Nodo(2)
cabeza'siguiente'siguiente := Nodo(3)

let cur := cabeza
repeat begin
  out(cur'valor)
  cur := cur'siguiente
end until is_null(cur)

Dereferenciación segura

is_null(x) comprueba si una referencia es Null antes de proyectar un campo. Proyectar sobre Null directamente produce un error claro y capturable (check/catch) en vez de un fallo interno.

Árbol binario

structure NodoArbol :=
  property valor
  property izq := Null
  property der := Null
  create(v) begin this'valor := v end

  where insertar(v) begin
    given v < this'valor:
      case T ->
        given is_null(this'izq):
          case T -> this'izq := NodoArbol(v)
          otherwise -> this'izq'insertar(v)
      otherwise ->
        given is_null(this'der):
          case T -> this'der := NodoArbol(v)
          otherwise -> this'der'insertar(v)
  end
end

Grafo (lista de adyacencia)

structure Grafo :=
  property adj := Mapping()
  where agregar_arista(a, b) begin
    given contains(this'adj, a):
      case F -> this'adj[a] := []
    given contains(this'adj, b):
      case F -> this'adj[b] := []
    this'adj[a] := this'adj[a] + [b]
    this'adj[b] := this'adj[b] + [a]
  end
end

contains(coleccion, elemento) prueba pertenencia en Sequence, Set o las llaves de un Mapping — Axioma no tiene un operador infijo in fuera de each y las comprensiones.

10Manejo de errores

check begin
  abort "algo salió mal"
end
catch Error as e begin
  out(s"capturado: {e}")
end
always begin
  out("limpieza ejecutada")
end

abort "texto" es de tipo genérico Error. Para excepciones propias y tipadas, usa una structure como "clase de error" — se captura por su nombre:

structure FueraDeRango :=
  property indice
  create(i) begin this'indice := i end
end

where obtener_seguro(seq, i) begin
  given i < 0 || i >= #seq:
    case T -> abort FueraDeRango(i)
  report seq[i]
end

check begin
  out(obtener_seguro([1,2,3], 10))
end
catch FueraDeRango as e begin
  out(s"índice {e'indice} fuera de rango")
end

verify <expresión> aborta si la expresión es falsa — útil como aserción.

11Especificación formal (EBNF)

program ::= statement*
statement ::= var_decl | if_stmt | func_def | each_stmt | repeat_stmt
| match_stmt | check_stmt | abort_stmt | struct_def
| "report" expr | "stop" | "next" | "skip" | "drop" IDENT
| "verify" expr | assign_stmt | expr_stmt
var_decl ::= "let" IDENT [ (":" type | "::" type) ] [ ":=" expr ]
if_stmt ::= "let" expr block ("given" expr block)* ["otherwise" block]
block ::= "begin" statement* "end"
func_def ::= "where" IDENT "(" params ")" [":" type] block
params ::= [ param ("," param)* ]
param ::= ["..."] IDENT [":" type] [":=" expr]
each_stmt ::= "each" IDENT "in" expr block
repeat_stmt ::= "repeat" block "until" expr
match_stmt ::= "given" expr ":" case_arm* [ "otherwise" "->" arm_body ]
case_arm ::= "case" expr "->" arm_body
arm_body ::= block | statement
check_stmt ::= "check" block catch_clause* ["resolve" block] ["always" block]
catch_clause ::= "catch" [IDENT] ["as" IDENT] block
abort_stmt ::= "abort" expr ["cause" expr]
struct_def ::= "structure" IDENT ["(" "immutable" ")"] ["extends" IDENT]
":=" struct_member* "end"
struct_member ::= ["static"] ["fixed"|"immutable"] "property" field_decl
| field_decl | func_def | "create" "(" params ")" block
field_decl ::= field_name [":" type] [":=" expr]
field_name ::= IDENT | KEYWORD -- permite nombres como 'next'
expr ::= ternary
ternary ::= or_expr ["?" or_expr ":" ternary]
or_expr ::= and_expr ("||" and_expr)*
and_expr ::= not_expr ("&&" not_expr)*
not_expr ::= "!!" not_expr | comparison
comparison ::= set_expr (("===" |"=!="|"<"|">"|"<="|">=") set_expr)*
set_expr ::= additive (("union"|"intersect"|"exclude"|"cart") additive)*
additive ::= multiplicative (("+"|"-") multiplicative)*
multiplicative ::= power (("*"|"/"|"//"|"%") power)*
power ::= unary ["**" power]
unary ::= "-" unary | "#" postfix | postfix
postfix ::= primary ( call | projection | index | slice )*
call ::= "(" [expr ("," expr)*] ")"
projection ::= "'" field_name
index ::= "[" expr "]"
slice ::= "[" expr ".." expr ["by" expr] "]"
primary ::= literal | IDENT | "(" expr ")" | list | set_or_map
| range | lambda | "fn" ...
lambda ::= "fn" "(" params ")" "=>" expr
list ::= "[" [expr ("," expr)* | expr "|" comp_clauses] "]"
set_or_map ::= "{" ( expr ".." expr -- rango
| expr ":" expr ("," expr ":" expr)* -- mapping
| expr ("," expr)* | expr "|" comp_clauses) "}"
comp_clauses ::= IDENT "in" expr ("," expr)* -- condiciones adicionales
